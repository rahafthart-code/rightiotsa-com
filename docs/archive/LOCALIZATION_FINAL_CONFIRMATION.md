**الملف**: `frontend/src/pages/UnifiedDashboard.jsx`
**السطور**: 150-169

```javascript
map.on('load', () => {
  // تحويل تسميات الدول للعربية
  map.setLayoutProperty('country-label', 'text-field', ['get', 'name_ar']);
  
  // تحويل تسميات المدن والمحافظات
  map.setLayoutProperty('state-label', 'text-field', ['get', 'name_ar']);
  map.setLayoutProperty('settlement-label', 'text-field', ['get', 'name_ar']);
  
  // مسح تلقائي لجميع طبقات التسميات وتحويلها للعربية
  const layers = map.getStyle().layers;
  layers.forEach(layer => {
    if (layer.id.includes('label')) {
      map.setLayoutProperty(layer.id, 'text-field', 
        ['coalesce', ['get', 'name_ar'], ['get', 'name']]
      );
    }
  });
});
```
